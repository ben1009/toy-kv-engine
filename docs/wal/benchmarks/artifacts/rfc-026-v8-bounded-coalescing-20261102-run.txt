import pathlib,subprocess,json,time,sys
base=pathlib.Path(__file__).resolve().parent
plan=json.loads((base/'plan.json').read_text())
for i,(threads,arm) in enumerate(plan['schedule']):
 name=f'{i:02}-{arm}-{threads}'
 print('START',name,flush=True)
 cmd=[str(base/('write-perf-'+arm)),'--bench','wal_batch_concurrent','--wal-batch-size','64','--pitr','--pitr-format','v8','--wal-io-mode','parallel','--num','750000','--threads',str(threads),'--value-size','1024','--target-sst-size','2147483648','--compaction','none','--output','json','--latency-sample-every','10','--path',str(base/(name+'-db'))]
 (base/(name+'.psi-before')).write_text(pathlib.Path('/proc/pressure/io').read_text())
 try:
  p=subprocess.run(cmd,text=True,capture_output=True,timeout=120)
  (base/(name+'.json')).write_text(p.stdout);(base/(name+'.stderr')).write_text(p.stderr)
  if p.returncode: print('FAILED',name,p.returncode,p.stderr[-500:],flush=True);continue
  d=json.loads(p.stdout); r=d['result'];w=r['wal_pipeline'];l=d['latency'];mean=w['fdatasync_ns']/w['sync_count']/1e6
  print(json.dumps({'name':name,'ops_s':round(r['ops_per_sec']),'p99_ms':l['p99_ms'],'max_ms':l['max_ms'],'syncs':w['sync_count'],'sync_mean_ms':round(mean,3),'cpu_s':round((r['process_cpu_user_ms']+r['process_cpu_system_ms'])/1000,3),'wait_ms':round(w['sync_coalesce_wait_ns']/1e6,2),'eventfd':w['worker_eventfd_writes'],'stall':l['p99_ms']>10 or mean>1}),flush=True)
 except subprocess.TimeoutExpired: print('TIMEOUT',name,flush=True)
 (base/(name+'.psi-after')).write_text(pathlib.Path('/proc/pressure/io').read_text())
 time.sleep(3)
