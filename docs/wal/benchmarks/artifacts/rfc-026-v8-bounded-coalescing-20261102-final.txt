import pathlib,subprocess,json,time
base=pathlib.Path(__file__).resolve().parent
schedule=[('batch',1,750000,'candidate-final'),('batch',1,750000,'baseline'),('batch',8,750000,'baseline'),('batch',8,750000,'candidate-final'),('batch',16,750000,'candidate-final'),('batch',16,750000,'baseline'),('point',1,5000,'baseline'),('point',1,5000,'candidate-final'),('point',8,20000,'candidate-final'),('point',8,20000,'baseline'),('point',16,20000,'baseline'),('point',16,20000,'candidate-final')]
(base/'final-plan.json').write_text(json.dumps({'schedule':schedule,'stall_flags':'p99>10ms or mean sync>1ms; retain all raw samples'},indent=2))
for i,(work,threads,num,arm) in enumerate(schedule):
 name=f'final-{i:02}-{work}-{arm}-{threads}'; print('START',name,flush=True)
 cmd=[str(base/('write-perf-'+arm)),'--bench','wal_batch_concurrent' if work=='batch' else 'wal_concurrent','--wal-batch-size','64','--pitr','--pitr-format','v8','--wal-io-mode','parallel','--num',str(num),'--threads',str(threads),'--value-size','1024','--target-sst-size','2147483648','--compaction','none','--output','json','--latency-sample-every','10','--path',str(base/(name+'-db'))]
 try:
  p=subprocess.run(cmd,text=True,capture_output=True,timeout=120);(base/(name+'.json')).write_text(p.stdout);(base/(name+'.stderr')).write_text(p.stderr)
  if p.returncode: print('FAILED',name,p.returncode,p.stderr[-500:],flush=True);continue
  d=json.loads(p.stdout);r=d['result'];w=r['wal_pipeline'];l=d['latency'];mean=w['fdatasync_ns']/w['sync_count']/1e6
  print(json.dumps({'name':name,'ops_s':round(r['ops_per_sec']),'p99_ms':l['p99_ms'],'max_ms':l['max_ms'],'syncs':w['sync_count'],'sync_mean_ms':round(mean,3),'cpu_s':round((r['process_cpu_user_ms']+r['process_cpu_system_ms'])/1000,3),'wait_ms':round(w['sync_coalesce_wait_ns']/1e6,2),'eventfd':w['worker_eventfd_writes'],'stall':l['p99_ms']>10 or mean>1}),flush=True)
 except subprocess.TimeoutExpired: print('TIMEOUT',name,flush=True)
 time.sleep(3)
